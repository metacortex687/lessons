abstract class DynArray<T> {

	//Конструктор 
	constructor(capacity=16, capacity_controller=MultiplicativeCapacityController())
	{
		this._capacity_controller = capacity_controller;
		this._capacity_controller.bind(this, capacity);
	}
	
	//Команды
	
	// предусловие:
	// индекс меньше или равен размеру массива и не меньше нуля
	// постусловие:
	// в массив добавлено новое значение или заменено существующее
	// если индекс был равен размеру массива, то значение добавляется
	// иначе заменяется
	// в некоторых случаях добавление происходит увеличение capacity(вместимости) массива
	public set(index: usize, valaue:T): void;
	
	
	// предусловие:
	// индекс внутри массива или на границе
	// постусловие:
	// в массиве удалено значение
	// у всех элементов после него, индекс уменьшен на 1
	// в некоторых случаях возможно уменьшение capacity(вместимости) массива 
	public  remove(index: usize): void;
	
	
	// предусловие:
	// индекс внутри массива или на границе
	// постусловие:
	// в массив добавлено значение
	// у всех элементов после него, индекс увеличен на 1
	// в некоторых случаях возможно увеличение вместимости массива
	public insert(index: usize): void;
	
	// предусловие:
	// индекс не пустой
	// постусловие:
	// удален последний элемент массива
	// возможно уменьшение вместимости массива
	public remove_last(): void;
	
	
	// постусловие:
	// в массив добавлен новый элемент
	// возможно увеличение вместимости массива
	public append(valaue:T): void;
	
	// постусловие:
	// изменена вместимость массива
	public resize(new_size:usize): void;
	
	//Запросы:
	
	//предусловие: массив не пуст	
	public get(index: usize): T;
	
	public size(): usize {
		return this._capacity_controller.size()
	}
	
	public capacity(): usize {
		return this._capacity_controller.capacity()		
	}
	
	//запросы статусов для команд и запросов имеющих предусловия
	...

}

А далее классы AdditiveCapacityController, MultiplicativeCapacityController, BankersMethodCapacityController наследуются от CapacityController и используют метод resize динамического массива для изменения вместимости.

В частично реализованном классе CapacityController будет реализован подсчет размера массива и его вместимости, что смогут использовать дочерние классы.

Методы класса CapacityController вида `on_*`, например `on_append`, должны вызываться вначале соответствующих методов динамического массива, например `append`.
