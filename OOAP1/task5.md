
```TS
abstract class Queue<T>
{
//Конструктор
//создает пустую очередь
public constructor();

//Команды:
//постусловие: в хвост очереди добавлен элемент
public enqueue(value: T): void;

//предусловие: очередь не пустая
//постусловие: из головы очереди удален элемент
public pop(): void;

//постусловие: 
// n элементов из головы очереди перемещены в хвост
//если очередь пустая, то без изменений
public rotate(n: uint): void;

//постусловие:
//очередь перевернута в другом направлении, те элементы что были в хвосте стали в голове
//пустая очередь без изменений
public reverse(): void;

//Запросы:

//предусловие: очередь не пустая
public peek(): T;

public size(): uint;

public is_empty(): bool; 

// запросы статусов (возможные значения статусов)
public get_pop_status(): int; // успешно; очередь пуста;
public get_peek_status(): int; // успешно; очередь пуста;

}
```
